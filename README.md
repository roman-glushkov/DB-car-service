## 🏗️ Архитектура проекта

Проект разделён на клиентскую и серверную части для удобства поддержки и масштабирования.

```text
DB-car-service/
│
├── server.js                          # Точка входа. Запуск сервера и подключение роутов.
├── .env                               # Настройки подключения к БД и секреты.
├── .env.example                       # Пример переменных окружения.
├── .gitignore                         # Исключения Git.
├── package.json                       # Зависимости и команды проекта.
├── package-lock.json                  # Зафиксированные версии зависимостей.
├── README.md                          # Документация проекта.
│
├── data/                              # Справочные данные приложения.
│   ├── dictionaries.json              # Словари и справочники.
│   ├── pages.json                     # Настройки и описания страниц.
│   └── status-colors.json             # Цвета статусов заказов.
│
├── src/                               # СЕРВЕРНАЯ ЛОГИКА (Backend)
│   │
│   ├── config/
│   │   ├── db.js                      # Пул подключений к MariaDB.
│   │   └── paths.js                   # Общие пути проекта.
│   │
│   ├── middleware/
│   │   ├── auth.js                    # Проверка сессий и ролей пользователей.
│   │   ├── car-upload.js              # Загрузка фотографий автомобилей через Multer.
│   │   └── error-handler.js           # Централизованная обработка ошибок.
│   │
│   ├── utils/
│   │   ├── passwords.js               # Хеширование и проверка паролей (scrypt).
│   │   └── validators.js              # Проверка и обработка входных данных.
│   │
│   ├── services/
│   │   └── order-service.js            # Бизнес-логика заказов и история изменений.
│   │
│   └── routes/                        # API-маршруты
│       ├── auth.js                    # Авторизация, выход, получение пользователя.
│       ├── client.js                  # API личного кабинета клиента.
│       ├── employee.js                # API личного кабинета мастера.
│       ├── admin.js                   # Подключение административных маршрутов.
│       │
│       └── admin/                     # API административной панели.
│           ├── accounts.js            # Управление учётными записями.
│           ├── cars.js                # Управление автомобилями.
│           ├── clients.js             # Управление клиентами.
│           ├── employees.js           # Управление сотрудниками.
│           ├── order-read.js          # Получение данных и списков заказов.
│           ├── order-write.js         # Создание и изменение заказов.
│           ├── orders.js              # Маршруты управления заказами.
│           ├── parts.js               # Управление запчастями.
│           ├── payments.js            # Управление оплатами.
│           ├── works.js               # Управление видами работ.
│
├── public/                            # КЛИЕНТСКАЯ ЧАСТЬ (Frontend)
│   │
│   ├── css/
│   │   ├── style.css                 # Стили административной панели.
│   │   ├── portal.css                # Стили кабинетов клиента и мастера.
│   │   └── login.css                 # Стили страницы авторизации.
│   │
│   ├── js/
│   │   ├── common/
│   │   │   └── api.js                # Общие API-запросы и вспомогательные функции.
│   │   │
│   │   ├── admin/                    # Модули административной панели.
│   │   │   ├── main.js               # Инициализация и маршрутизация админки.
│   │   │   ├── ui.js                 # Общие элементы интерфейса.
│   │   │   ├── filters.js            # Фильтрация и поиск.
│   │   │   ├── order-modal.js        # Модальные окна заказов.
│   │   │   ├── orders.js             # Логика заказов.
│   │   │   ├── clients.js            # Логика клиентов.
│   │   │   ├── cars.js               # Логика автомобилей.
│   │   │   ├── employees.js          # Логика сотрудников.
│   │   │   ├── works.js              # Логика работ.
│   │   │   ├── parts.js              # Логика запчастей.
│   │   │   ├── payments.js           # Логика оплат.
│   │   │   ├── accounts.js           # Логика учётных записей.
│   │   │   └── settings.js           # Настройки административной панели.
│   │   │
│   │   ├── client.js                 # Логика кабинета клиента.
│   │   ├── employee.js               # Логика кабинета мастера.
│   │   └── login.js                  # Логика страницы авторизации.
│   │
│   ├── admin.html                    # Страница административной панели.
│   ├── client.html                   # Личный кабинет клиента.
│   ├── employee.html                 # Личный кабинет мастера.
│   └── login.html                    # Страница авторизации.
│
└── uploads/                          # Файловое хранилище.
    └── cars/                         # Загруженные фотографии автомобилей.
        ├── .gitkeep
        └── *.webp                    # Фотографии автомобилей.
```
